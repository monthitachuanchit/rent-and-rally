    // Simple data for racquets

    const racquets = [

      { id: 1, name: "Wilson Pro Staff 97", level: "Advanced", pricePerDay: 12, info: "Control‑focused, for experienced players." },

      { id: 2, name: "Babolat Pure Drive", level: "Intermediate", pricePerDay: 10, info: "Great power and spin for club players." },

      { id: 3, name: "Head Radical Team", level: "Beginner", pricePerDay: 8, info: "Lightweight and easy to play with." }

    ];


    const racquetList = document.getElementById("racquetList");

    const racquetSelect = document.getElementById("racquetSelect");


    // Show racquets on page

    racquets.forEach(r => {

      const card = document.createElement("div");

      card.className = "card";

      card.innerHTML = `

        <h3>${r.name}</h3>

        <div>Level: ${r.level}</div>

        <div>${r.info}</div>

        <div class="price">$${r.pricePerDay} / day</div>

      `;

      racquetList.appendChild(card);


      const opt = document.createElement("option");

      opt.value = r.id;

      opt.textContent = `${r.name} (${r.level})`;

      racquetSelect.appendChild(opt);

    });


    // Booking form (front‑end only)

    const bookingForm = document.getElementById("bookingForm");

    const formMessage = document.getElementById("formMessage");


    bookingForm.addEventListener("submit", function(e) {

      e.preventDefault();

      formMessage.textContent = "";

      formMessage.className = "message";


      const name = document.getElementById("name").value.trim();

      const email = document.getElementById("email").value.trim();

      const start = document.getElementById("start").value;

      const end = document.getElementById("end").value;


      if (!name || !email || !start || !end) {

        formMessage.textContent = "Please fill in all required fields.";

        formMessage.classList.add("error");

        return;

      }

      if (end < start) {

        formMessage.textContent = "End date must be after start date.";

        formMessage.classList.add("error");

        return;

      }


      formMessage.textContent = "Booking request recorded (demo only). In a real site this would be sent to a server.";

      formMessage.classList.add("success");

      bookingForm.reset();

      racquetSelect.selectedIndex = 0;

    });

  </script>
